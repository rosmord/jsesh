		}
	}

	public boolean getBooleanModifier(String name) {
		Modifier m= getModifierFromName(name, NO_VALUE);
		return (m != null);
	}
	
	public void setIntModifier(String name, Integer value) {
		Modifier m= getModifierFromName(name, VALUE);
		if (m != null) {
			m.setValue(value);
		} else
			modifiers.add(new Modifier(name, value));
	} // end setModifier
